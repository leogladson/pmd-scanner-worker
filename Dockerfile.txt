
FROM node:22-bookworm

# Install Java and system dependencies required by the scanner
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
      openjdk-17-jdk-headless \
      python3 \
      git \
      ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Install Salesforce CLI and Code Analyzer
RUN npm install -g @salesforce/cli && \
    sf plugins install code-analyzer

WORKDIR /app

# Install reproducible Node.js dependencies
COPY package.json package-lock.json ./
RUN npm ci --omit=dev

# Copy application source, configuration and scanner rules
COPY . .

# Create the workspace used for scans
RUN mkdir -p /app/workspace

ENV NODE_ENV=production
ENV PORT=10000
ENV SCAN_WORKSPACE=/app/workspace

EXPOSE 10000

CMD ["npm", "start"]
